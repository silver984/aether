add_library(aether_engine_core STATIC)

file(GLOB_RECURSE AE_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/include/aether/*.h*)
file(GLOB AE_EXCLUDED_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/include/aether/excluded/*.h*)
list(REMOVE_ITEM AE_HEADERS ${AE_EXCLUDED_HEADERS})
file(GLOB_RECURSE AE_EXTERNAL_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/external/*.h*)

file(GLOB AE_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c*)
file(GLOB AE_NODE_COMP_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/node/components/*.c*)

target_sources(
  aether_engine_core
  PUBLIC
    FILE_SET HEADERS
    BASE_DIRS
      ${CMAKE_CURRENT_SOURCE_DIR}/include
      ${CMAKE_CURRENT_SOURCE_DIR}/external # maybe todo: make this private?
    FILES
      ${AE_HEADERS}
      ${AE_EXTERNAL_HEADERS}
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/node/node.cc
    ${AE_SRC}
    ${AE_NODE_COMP_SRC}
)

add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/external)

include(${CMAKE_SOURCE_DIR}/cmake/CPM.cmake)

  NAME raylib
  GITHUB_REPOSITORY raysan5/raylib

    SUPPORT_FILEFORMAT_MTL=0
    SUPPORT_FILEFORMAT_IQM=0
    SUPPORT_FILEFORMAT_GLTF=0
    SUPPORT_FILEFORMAT_VOX=0
    SUPPORT_FILEFORMAT_M3D=0
    SUPPORT_MESH_GENERATION=0
)

CPMAddPackage(
  NAME lua_src
  GITHUB_REPOSITORY lua/lua
  GIT_TAG v5.4.8
  GIT_SHALLOW TRUE
)

file(GLOB LUA_SRC ${lua_src_SOURCE_DIR}/*.c)

list(
  REMOVE_ITEM LUA_SRC
    ${lua_src_SOURCE_DIR}/lua.c
    ${lua_src_SOURCE_DIR}/onelua.c
)

add_library(lua STATIC ${LUA_SRC})
target_include_directories(lua PUBLIC ${lua_src_SOURCE_DIR})
